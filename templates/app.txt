@app.route('/api/cart/add', methods=['POST'])
def add_to_cart():
    try:
        data = request.get_json()
        if not data or 'product_id' not in data:
            return jsonify({'success': False, 'error': 'Invalid request payload'}), 400
        product_id = int(data['product_id'])
        quantity = int(data.get('quantity', 1))
        if quantity < 1:
            return jsonify({'success': False, 'error': 'Quantity must be at least 1'}), 400
        product = next((p for p in PRODUCTS if p['id'] == product_id), None)
        if not product:
            return jsonify({'success': False, 'error': 'Product not found in catalog'}), 404
        cart = session.get('cart', [])
        for item in cart:
            if item['id'] == product_id:
                item['quantity'] += quantity
                break
        else:
            cart.append({
                'id': product_id,
                'name': product['name'],
                'price': product['price'],
                'quantity': quantity,
                'image': product['image']
            })
        session['cart'] = cart
        session.modified = True
        cart_dict = {item['id']: item['quantity'] for item in cart}
        distinct_count = len(cart)
        return jsonify({
            'success': True,
            'cart_count': distinct_count,
            'cart_dict': cart_dict,
            'message': f"{product['name']} added to cart!",
            'quantity': next(item['quantity'] for item in cart if item['id'] == product_id)
        })
    except (ValueError, TypeError):
        return jsonify({'success': False, 'error': 'Invalid data format'}), 400
    except Exception as e:
        app.logger.error(f"Cart add error: {str(e)}")
        return jsonify({'success': False, 'error': 'Internal server error'}), 500

@app.route('/api/cart/update', methods=['POST'])
def update_cart():
    try:
        data = request.get_json()
        if not data or 'product_id' not in data or 'quantity' not in data:
            return jsonify({'success': False, 'error': 'Missing required fields'}), 400
        product_id = int(data['product_id'])
        new_quantity = int(data['quantity'])
        if new_quantity < 0:
            return jsonify({'success': False, 'error': 'Invalid quantity value'}), 400
        cart = session.get('cart', [])
        item_found = False
        if new_quantity == 0:
            original_length = len(cart)
            cart = [item for item in cart if item['id'] != product_id]
            item_found = (len(cart) < original_length)
        else:
            for item in cart:
                if item['id'] == product_id:
                    item['quantity'] = new_quantity
                    item_found = True
                    break
        if not item_found:
            product = next((p for p in PRODUCTS if p['id'] == product_id), None)
            if not product:
                return jsonify({'success': False, 'error': 'Product not found'}), 404
            cart.append({
                'id': product_id,
                'name': product['name'],
                'price': product['price'],
                'quantity': new_quantity,
                'image': product['image']
            })
            item_found = True
        session['cart'] = cart
        session.modified = True
        cart_dict = {item['id']: item['quantity'] for item in cart}
        distinct_count = len(cart)
        subtotal = sum(item['price'] * item['quantity'] for item in cart)
        item_total = next((item['price'] * item['quantity'] for item in cart if item['id'] == product_id), 0) if new_quantity > 0 else 0
        return jsonify({
            'success': True,
            'cart_count': distinct_count,
            'cart_dict': cart_dict,
            'subtotal': subtotal,
            'item_total': item_total,
            'quantity': new_quantity if new_quantity > 0 else 0
        })
    except (ValueError, TypeError):
        return jsonify({'success': False, 'error': 'Invalid data format'}), 400
    except Exception as e:
        app.logger.error(f"Cart update error: {str(e)}")
        return jsonify({'success': False, 'error': 'Internal server error'}), 500

@app.route('/api/coupon/validate', methods=['POST'])
def validate_coupon():
    try:
        data = request.get_json()
        code = data.get('code', '').strip().upper()
        subtotal = float(data.get('subtotal', 0))
        if not code:
            return jsonify({'success': False, 'error': 'Coupon code is required'}), 400
        if code in COUPONS.get('50_percent', []):
            discount_percent = 50
        elif code in COUPONS.get('25_percent', []):
            discount_percent = 25
        else:
            return jsonify({'success': False, 'error': 'Invalid or expired coupon'}), 400
        discount_amount = round(subtotal * (discount_percent / 100), 2)
        final_amount = subtotal - discount_amount
        return jsonify({
            'success': True,
            'discount_percent': discount_percent,
            'discount_amount': discount_amount,
            'final_amount': final_amount,
            'message': f'{discount_percent}% discount applied!'
        })
    except (ValueError, TypeError):
        return jsonify({'success': False, 'error': 'Invalid subtotal value'}), 400
    except Exception as e:
        app.logger.error(f"Coupon validation error: {str(e)}")
        return jsonify({'success': False, 'error': 'Internal server error'}), 500

class GlobalCartManager {
constructor() {
this.cartBadge = document.getElementById('cart-badge');
this.cartLink = document.getElementById('cart-link');
this.cartState = {}; // { productId: quantity }
this.syncing = false;
this.init();
}
async init() {
// ALWAYS fetch fresh state on load (bypass cache)
await this.refreshCartState();

// ✅ CRITICAL FIX: Dispatch globalCartSync AFTER initial load
document.dispatchEvent(new CustomEvent('globalCartSync', {
detail: {
cartState: { ...this.cartState },
cartCount: Object.keys(this.cartState).length,
source: 'initial_load'
}
}));

// Listen for cart operations from ANY component
document.addEventListener('cartAction', async (e) => {
if (this.syncing) return;
const { productId, quantity, action } = e.detail;
try {
this.syncing = true;
let endpoint, payload;
if (action === 'add') {
endpoint = '/api/cart/add';
payload = { product_id: productId, quantity: quantity || 1 };
} else if (action === 'update') {
endpoint = '/api/cart/update';
payload = { product_id: productId, quantity: quantity };
} else if (action === 'clear') {
endpoint = '/api/cart/clear';
payload = {};
} else {
throw new Error('Invalid cart action');
}
const response = await fetch(endpoint, {
method: 'POST',
headers: {
'Content-Type': 'application/json',
'X-Requested-With': 'XMLHttpRequest'
},
body: JSON.stringify(payload),
credentials: 'same-origin'
});
if (!response.ok) {
const err = await response.json().catch(() => ({}));
throw new Error(err.error || `HTTP ${response.status}`);
}
const data = await response.json();
if (!data.success) throw new Error(data.error || 'Operation failed');
// FULL STATE REFRESH after mutation
await this.refreshCartState();
// Dispatch global sync event
document.dispatchEvent(new CustomEvent('globalCartSync', {
detail: {
cartState: { ...this.cartState },
cartCount: data.cart_count || 0,
source: action
}
}));
// Show toast if available
if (data.message) this.showToast(data.message, 'success');
} catch (error) {
console.error('Cart action failed:', error);
this.showToast(error.message || 'Update failed. Please try again.', 'error');
} finally {
this.syncing = false;
}
});

// Handle page visibility changes (tab switch/back button)
document.addEventListener('visibilitychange', () => {
if (document.visibilityState === 'visible' && !this.syncing) {
this.refreshCartState().then(() => {
document.dispatchEvent(new CustomEvent('globalCartSync', {
detail: {
cartState: { ...this.cartState },
cartCount: Object.keys(this.cartState).length,
source: 'visibility_change'
}
}));
});
}
});
}
async refreshCartState() {
try {
// Bypass cache with timestamp + cache-control headers
const response = await fetch(`/api/cart/state?_t=${Date.now()}`, {
headers: {
'Cache-Control': 'no-cache, no-store, must-revalidate',
'Pragma': 'no-cache',
'Expires': '0'
},
credentials: 'same-origin'
});
if (!response.ok) throw new Error(`HTTP ${response.status}`);
const data = await response.json();
if (!data.success) throw new Error('Invalid cart state response');
// Update local state
this.cartState = data.cart_dict || {};
this.updateBadge(data.distinct_count || 0);
} catch (error) {
console.warn('Failed to refresh cart state:', error);
// Don't show toast on init failures to avoid spam
if (!document.documentElement.classList.contains('cart-initialized')) {
document.documentElement.classList.add('cart-initialized');
}
}
}
updateBadge(count) {
if (count > 0) {
if (!this.cartBadge) {

document.addEventListener('DOMContentLoaded', () => {
  const handleCartUpdate = async (productId, newQuantity, actionName) => {
    try {
      const response = await fetch('/api/cart/update', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Requested-With': 'XMLHttpRequest'
        },
        body: JSON.stringify({
          product_id: productId,
          quantity: newQuantity
        }),
        credentials: 'same-origin'
      });
      if (!response.ok) {
        const err = await response.json().catch(() => ({}));
        throw new Error(err.error || `HTTP ${response.status}`);
      }
      const data = await response.json();
      if (!data.success) throw new Error(data.error || 'Update failed');
      window.location.reload();
    } catch (error) {
      console.error(`${actionName} failed:`, error);
      if (window.globalCart?.showToast) {
        window.globalCart.showToast(error.message || `Failed to ${actionName.toLowerCase()}`, 'error');
      }
    }
  };

  // Quantity controls (mobile + desktop)
  document.querySelectorAll('.quantity-dec, .quantity-inc').forEach(button => {
    button.addEventListener('click', (e) => {
      e.preventDefault();
      e.stopPropagation();
      const productId = parseInt(button.closest('[data-product-id]').dataset.productId);
      const container = button.closest('.cart-item');
      const qtySpan = container.querySelector('.cart-quantity');
      const currentQty = parseInt(qtySpan.textContent);
      const isIncrement = button.classList.contains('quantity-inc');
      const newQty = isIncrement ? currentQty + 1 : Math.max(0, currentQty - 1);

      if (newQty === 0) {
        if (!confirm('Remove this item from cart?')) return;
        handleCartUpdate(productId, 0, 'Remove item');
      } else {
        handleCartUpdate(productId, newQty, isIncrement ? 'Increase quantity' : 'Decrease quantity');
      }
    });
  });

  // Remove item
  document.querySelectorAll('.remove-item').forEach(button => {
    button.addEventListener('click', (e) => {
      e.preventDefault();
      e.stopPropagation();
      if (!confirm('Remove this item from cart?')) return;
      const productId = parseInt(button.closest('[data-product-id]').dataset.productId);
      handleCartUpdate(productId, 0, 'Remove item');
    });
  });

  // Clear cart
  document.getElementById('clear-cart-btn')?.addEventListener('click', async (e) => {
    e.preventDefault();
    if (!confirm('Clear your entire cart? This cannot be undone.')) return;
    try {
      const response = await fetch('/api/cart/clear', {
        method: 'POST',
        headers: { 'X-Requested-With': 'XMLHttpRequest' },
        credentials: 'same-origin'
      });
      if (!response.ok) throw new Error(`HTTP ${response.status}`);
      const data = await response.json();
      if (!data.success) throw new Error(data.error || 'Clear failed');
      window.location.reload();
    } catch (error) {
      console.error('Clear cart failed:', error);
      if (window.globalCart?.showToast) {
        window.globalCart.showToast(error.message || 'Failed to clear cart', 'error');
      }
    }
  });
});

<script>
document.addEventListener('DOMContentLoaded', () => {
  const productId = {{ product.id }};
  const actionContainer = document.getElementById('product-action-container');

  // Reusable function to render "Add to Cart" button
  function renderAddButton() {
    actionContainer.innerHTML = `
      <button id="add-to-cart-btn" class="w-full bg-primary hover:bg-indigo-700 text-white font-bold py-3 px-6 rounded-xl text-lg transition flex items-center justify-center shadow-md hover:shadow-lg">
        <i class="fas fa-cart-plus mr-2"></i> Add to Cart
      </button>
      <a href="/products" class="mt-3 w-full text-center py-2 text-gray-600 hover:text-primary font-medium transition">
        ← Continue Shopping
      </a>
    `;
    document.getElementById('add-to-cart-btn').addEventListener('click', handleAddToCart);
  }

  // Reusable function to render quantity controls
  function renderQuantityControls(initialQty = 1) {
    let qty = initialQty;
    actionContainer.innerHTML = `
      <div class="flex items-center justify-between max-w-xs mx-auto w-full">
        <span class="text-gray-700 font-medium">In Cart:</span>
        <div class="flex items-center border rounded-lg overflow-hidden">
          <button id="dec-btn" class="px-3 py-2 hover:bg-gray-100 ${qty <= 1 ? 'disabled:opacity-50' : ''}" ${qty <= 1 ? 'disabled' : ''}>
            <i class="fas fa-minus text-gray-600"></i>
          </button>
          <span id="quantity-display" class="px-4 py-2 font-bold">${qty}</span>
          <button id="inc-btn" class="px-3 py-2 hover:bg-gray-100">
            <i class="fas fa-plus text-primary"></i>
          </button>
        </div>
      </div>
      <a href="/cart" class="mt-4 w-full bg-green-500 hover:bg-green-600 text-white font-bold py-3 rounded-xl text-center transition flex items-center justify-center">
        <i class="fas fa-shopping-cart mr-2"></i> Go to Cart
      </a>
      <a href="/" class="mt-3 w-full text-center py-2 text-gray-600 hover:text-primary font-medium transition">
        ← Continue Shopping
      </a>
    `;

    // Attach event listeners
    document.getElementById('dec-btn').addEventListener('click', () => {
      if (qty > 1) {
        qty--;
        document.getElementById('quantity-display').textContent = qty;
        const decBtn = document.getElementById('dec-btn');
        decBtn.disabled = (qty <= 1);
        decBtn.classList.toggle('disabled:opacity-50', qty <= 1);
        window.globalCart.updateItemQuantity(productId, qty);
      } else {
        // Remove from cart and revert to "Add" button
        window.globalCart.updateItemQuantity(productId, 0).finally(() => {
          renderAddButton();
        });
      }
    });

    document.getElementById('inc-btn').addEventListener('click', () => {
      qty++;
      document.getElementById('quantity-display').textContent = qty;
      document.getElementById('dec-btn').disabled = false;
      document.getElementById('dec-btn').classList.remove('disabled:opacity-50');
      window.globalCart.updateItemQuantity(productId, qty);
    });
  }

  // Handle "Add to Cart" click
  async function handleAddToCart() {
    const btn = document.getElementById('add-to-cart-btn');
    if (!btn) return;

    const originalHTML = btn.innerHTML;
    btn.innerHTML = '<i class="fas fa-spinner fa-spin mr-2"></i> Adding...';
    btn.disabled = true;

    try {
      // Wait for cart update to complete
      await window.globalCart.addToCart(productId, 1);
      // Only render controls AFTER success
      renderQuantityControls(1);
    } catch (error) {
      console.error('Add to cart failed:', error);
      // Restore button on error
      btn.innerHTML = originalHTML;
      btn.disabled = false;
      if (window.globalCart?.showToast) {
        window.globalCart.showToast('Failed to add item. Please try again.', 'error');
      }
    }
  }

  // Initialize based on initial state
  {% if in_cart %}
    renderQuantityControls({{ cart_quantity }});
  {% else %}
    document.getElementById('add-to-cart-btn')?.addEventListener('click', handleAddToCart);
  {% endif %}
});
</script>
{% endblock %}
